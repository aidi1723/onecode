### 任务 2.1: 纳甲地支映射

#### 目标
为每个 6-bit 状态码的六爻分配对应的地支序列。

#### 实现方案
```python
# 文件: src/onecode/kernel/hexagram.py
# 位置: IchingKernel 类中新增方法

@staticmethod
def hexagram_earthly_branches(status_code: int) -> list[str]:
    """
    返回六爻对应的地支序列（从初爻到上爻）
    
    Args:
        status_code: 6-bit 状态码 (0-63)
        
    Returns:
        地支列表，例如 ["子", "寅", "辰", "午", "申", "戌"]
        
    用途:
        - 可视化时显示地支标注
        - 季节分析的基础数据
        - 方位分析的参考维度
    """
    inner = status_code & 0b111
    outer = (status_code >> 3) & 0b111
    
    # 八卦纳甲地支表（按照传统口诀）
    TRIGRAM_BRANCHES = {
        0b111: ["子", "寅", "辰", "午", "申", "戌"],  # 乾(内甲子外壬午)
        0b010: ["寅", "辰", "午", "申", "戌", "子"],  # 坎(内戊寅外戊申)
        0b001: ["子", "寅", "辰", "午", "申", "戌"],  # 震(内庚子外庚午)
        0b100: ["辰", "午", "申", "戌", "子", "寅"],  # 艮(内丙辰外丙戌)
        0b000: ["未", "巳", "卯", "丑", "亥", "酉"],  # 坤(内乙未外癸丑)
        0b110: ["丑", "亥", "酉", "未", "巳", "卯"],  # 巽(内辛丑外辛未)
        0b101: ["卯", "丑", "亥", "酉", "未", "巳"],  # 离(内己卯外己酉)
        0b011: ["巳", "卯", "丑", "亥", "酉", "未"],  # 兑(内丁巳外丁亥)
    }
    
    # 内卦前三爻，外卦后三爻
    inner_branches = TRIGRAM_BRANCHES[inner][:3]
    outer_branches = TRIGRAM_BRANCHES[outer][3:]
    return inner_branches + outer_branches
```

### 任务 2.2: 卦宫归属算法（安世应）

#### 目标
实现安世应算法，确定每个状态码归属的卦宫和世应位置。

#### 实现方案
```python
# 文件: src/onecode/kernel/hexagram.py
# 位置: IchingKernel 类中新增方法

@staticmethod
def palace_attribution(status_code: int) -> dict:
    """
    计算卦宫归属和世应位置（安世应算法）
    
    Args:
        status_code: 6-bit 状态码 (0-63)
        
    Returns:
        {
            "palace": 卦宫三卦码 (0-7),
            "palace_name": 卦宫名称 ("qian"/"kun"等),
            "palace_element": 卦宫五行 ("metal"/"earth"等),
            "world_line": 世爻位置 (0-5),
            "response_line": 应爻位置 (0-5),
            "hexagram_type": 卦型 ("pure"/"travel"/"return")
        }
        
    算法:
        1. 八纯卦（内外卦相同）→ 世爻在五爻（索引5）
        2. 非八纯卦: 从初爻开始逐爻阴阳互变
        3. 直到内外卦变成相同 → 该爻为世爻，该卦为卦宫
        4. 变到五爻仍不同 → 转向四爻往下变
        5. 变到初爻才相同 → 归魂卦，世爻在三爻（索引2）
        6. 应爻 = (世爻 + 3) % 6
    """
    inner = status_code & 0b111
    outer = (status_code >> 3) & 0b111
    
    # 八纯卦判断
    if inner == outer:
        return {
            "palace": inner,
            "palace_name": IchingKernel.TRIGRAM_NAMES[inner],
            "palace_element": IchingKernel.TRIGRAM_ELEMENTS[inner],
            "world_line": 5,  # 上爻
            "response_line": 2,  # 三爻
            "hexagram_type": "pure"
        }
    
    # 非八纯卦：逐爻阴阳互变寻找卦宫
    def flip_line(trigram, line_idx):
        """翻转三卦的某一爻"""
        return trigram ^ (1 << line_idx)
    
    # 从初爻开始向上变（0→1→2→3→4→5）
    current_inner = inner
    for world_line in range(6):
        if world_line < 3:
            # 变内卦
            test_inner = flip_line(current_inner, world_line)
            test_outer = outer
        else:
            # 变外卦
            test_inner = current_inner
            test_outer = flip_line(outer, world_line - 3)
        
        if test_inner == test_outer:
            # 找到卦宫
            return {
                "palace": test_inner,
                "palace_name": IchingKernel.TRIGRAM_NAMES[test_inner],
                "palace_element": IchingKernel.TRIGRAM_ELEMENTS[test_inner],
                "world_line": world_line,
                "response_line": (world_line + 3) % 6,
                "hexagram_type": "travel" if world_line != 2 else "return"
            }
    
    # 理论上不会到达这里（64卦都有归属）
    raise ValueError(f"Cannot find palace for status {status_code}")
```

### 任务 2.3: 六亲关系网

#### 目标
基于卦宫五行，为每一爻装配六亲关系。

#### 实现方案
```python
# 文件: src/onecode/kernel/hexagram.py
# 位置: IchingKernel 类中新增方法

@staticmethod
def six_relatives_profile(status_code: int) -> dict:
    """
    装配六亲关系网（基于卦宫五行）
    
    Args:
        status_code: 6-bit 状态码 (0-63)
        
    Returns:
        {
            "palace": 卦宫三卦码,
            "palace_element": 卦宫五行,
            "lines": [
                {
                    "line_index": 爻位索引 (0-5),
                    "earthly_branch": 地支,
                    "element": 地支五行,
                    "relative": 六亲关系 ("brother"/"parent"/"offspring"/"wealth"/"officer")
                },
                ...
            ]
        }
        
    六亲规则:
        - 与卦宫同五行 → 兄弟 (brother)
        - 生卦宫五行 → 父母 (parent)
        - 卦宫生之 → 子孙 (offspring)
        - 卦宫克之 → 妻财 (wealth)
        - 克卦宫 → 官鬼 (officer)
    """
    # 1. 获取卦宫归属
    palace_info = IchingKernel.palace_attribution(status_code)
    palace_element = palace_info["palace_element"]
    
    # 2. 获取地支序列
    branches = IchingKernel.hexagram_earthly_branches(status_code)
    
    # 3. 地支配五行表
    BRANCH_ELEMENTS = {
        "子": "water", "亥": "water",
        "丑": "earth", "辰": "earth", "未": "earth", "戌": "earth",
        "寅": "wood", "卯": "wood",
        "巳": "fire", "午": "fire",
        "申": "metal", "酉": "metal",
    }
    
    # 4. 为每一爻装六亲
    lines = []
    for i, branch in enumerate(branches):
        branch_element = BRANCH_ELEMENTS[branch]
        
        # 判断六亲关系
        if branch_element == palace_element:
            relative = "brother"
        elif IchingKernel._element_generates(branch_element, palace_element):
            relative = "parent"
        elif IchingKernel._element_generates(palace_element, branch_element):
            relative = "offspring"
        elif IchingKernel._element_controls(palace_element, branch_element):
            relative = "wealth"
        elif IchingKernel._element_controls(branch_element, palace_element):
            relative = "officer"
        else:
            relative = "unknown"  # 理论上不会出现
        
        lines.append({
            "line_index": i,
            "earthly_branch": branch,
            "element": branch_element,
            "relative": relative
        })
    
    return {
        "palace": palace_info["palace"],
        "palace_element": palace_element,
        "lines": lines
    }

@staticmethod
def _element_generates(element_a: str, element_b: str) -> bool:
    """判断五行相生关系"""
    generation_cycle = {
        "wood": "fire",
        "fire": "earth",
        "earth": "metal",
        "metal": "water",
        "water": "wood"
    }
    return generation_cycle.get(element_a) == element_b

@staticmethod
def _element_controls(element_a: str, element_b: str) -> bool:
    """判断五行相克关系"""
    control_cycle = {
        "wood": "earth",
        "earth": "water",
        "water": "fire",
        "fire": "metal",
        "metal": "wood"
    }
    return control_cycle.get(element_a) == element_b
```

#### 测试用例
```python
def test_six_relatives_profile_for_离_hexagram(self):
    """测试离为火卦的六亲关系"""
    # 离为火（42 = 0b101101）
    relatives = IchingKernel.six_relatives_profile(42)
    self.assertEqual(relatives["palace_element"], "fire")
    
    # 验证每一爻都有明确六亲
    for line in relatives["lines"]:
        self.assertIn(line["relative"], 
                     ["brother", "parent", "offspring", "wealth", "officer"])

def test_six_relatives_all_64_hexagrams_have_valid_relations(self):
    """测试所有64卦的六亲关系合法性"""
    for i in range(64):
        relatives = IchingKernel.six_relatives_profile(i)
        self.assertEqual(len(relatives["lines"]), 6)
        
        # 统计六亲分布（不同卦宫会有不同分布）
        relative_counts = {}
        for line in relatives["lines"]:
            rel = line["relative"]
            relative_counts[rel] = relative_counts.get(rel, 0) + 1
        
        # 至少包含兄弟（同五行必然存在）
        self.assertGreater(relative_counts.get("brother", 0), 0)
```
